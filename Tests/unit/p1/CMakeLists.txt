cmake_minimum_required(VERSION 3.15)
project(ThunderP1TransportTests LANGUAGES C CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(Threads REQUIRED)
set(HIBERNATE "${CMAKE_CURRENT_LIST_DIR}/../../../Source/addons/hibernate")
add_executable(p1_checkpoint_server test_checkpoint.cpp "${HIBERNATE}/checkpointserver/CheckpointServer.c")
target_compile_features(p1_checkpoint_server PRIVATE cxx_std_17)
target_link_libraries(p1_checkpoint_server PRIVATE GTest::gtest_main Threads::Threads)
target_link_options(p1_checkpoint_server PRIVATE
    "-Wl,--wrap=poll" "-Wl,--wrap=send" "-Wl,--wrap=recv" "-Wl,--wrap=__recv_chk")
add_test(NAME p1_checkpoint_server COMMAND p1_checkpoint_server)
set_tests_properties(p1_checkpoint_server PROPERTIES TIMEOUT 10)
add_executable(p1_checkpoint_library test_checkpoint_library.cpp "${HIBERNATE}/checkpointlib/CheckpointLib.c")
target_compile_features(p1_checkpoint_library PRIVATE cxx_std_17)
target_link_libraries(p1_checkpoint_library PRIVATE GTest::gtest_main)
add_test(NAME p1_checkpoint_library COMMAND p1_checkpoint_library)
add_executable(p1_security_path test_security_path.cpp)
target_compile_features(p1_security_path PRIVATE cxx_std_17)
target_link_libraries(p1_security_path PRIVATE GTest::gtest_main)
add_test(NAME p1_security_path COMMAND p1_security_path)
add_executable(p1_process_storage test_process_storage.cpp)
target_compile_features(p1_process_storage PRIVATE cxx_std_17)
target_link_libraries(p1_process_storage PRIVATE GTest::gtest_main)
add_test(NAME p1_process_storage COMMAND p1_process_storage)
if(TARGET ThunderCore)
    add_executable(p1_core
        ../core/test_frametype.cpp ../core/test_dataelement.cpp
        ../core/test_dataelementfile.cpp ../core/test_filesystem.cpp
        ../core/test_cryptalgo.cpp ../core/test_jwt.cpp ../core/test_p1_metrics.cpp)
    target_compile_features(p1_core PRIVATE cxx_std_17)
    target_include_directories(p1_core PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../Source/addons")
    target_link_libraries(p1_core PRIVATE ThunderCore ThunderCryptalgo
        ThunderWebSocket GTest::gtest_main Threads::Threads)
    if(SECURE_SOCKET)
        find_package(OpenSSL REQUIRED)
        target_link_libraries(p1_core PRIVATE OpenSSL::SSL OpenSSL::Crypto)
    endif()
    add_test(NAME p1_core COMMAND p1_core
        "--gtest_filter=test_frame.*:test_asn1_p1.*:test_file_p1.*:test_datafile_p1.*:URL_P1.*:TLS_P1.*:Cryptalgo_AES.*:JWT_*.*:CGroupP1.*")
    add_executable(p1_faults test_faults.cpp "${CMAKE_CURRENT_LIST_DIR}/../../../Source/cryptalgo/Random.cpp")
    target_compile_features(p1_faults PRIVATE cxx_std_17)
    target_compile_definitions(p1_faults PRIVATE MODULE_NAME=P1FaultTests)
    target_compile_options(p1_faults PRIVATE -fno-builtin-malloc -fno-builtin-realloc)
    target_link_libraries(p1_faults PRIVATE ThunderCore GTest::gtest_main)
    target_link_options(p1_faults PRIVATE
        "-Wl,--wrap=malloc" "-Wl,--wrap=realloc" "-Wl,--wrap=open" "-Wl,--wrap=read")
    add_test(NAME p1_faults COMMAND p1_faults)
    if(SECURE_SOCKET)
        find_package(OpenSSL REQUIRED)
        add_executable(p1_tls test_tls.cpp)
        target_compile_features(p1_tls PRIVATE cxx_std_17)
        target_compile_definitions(p1_tls PRIVATE MODULE_NAME=P1TlsTests)
        target_link_libraries(p1_tls PRIVATE ThunderCryptalgo ThunderCore
            GTest::gtest_main Threads::Threads OpenSSL::SSL OpenSSL::Crypto)
        add_test(NAME p1_tls COMMAND p1_tls "--gtest_filter=TLSHandshakeP1.*")
        set_tests_properties(p1_tls PROPERTIES TIMEOUT 30)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        find_program(P1_OPENSSL_EXECUTABLE NAMES openssl REQUIRED)
        foreach(source bundle directory)
            add_test(NAME p1_tls_default_${source}
                COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/run_default_trust.py
                    --openssl ${P1_OPENSSL_EXECUTABLE} --source ${source}
                    --test $<TARGET_FILE:p1_tls>)
            set_tests_properties(p1_tls_default_${source} PROPERTIES TIMEOUT 30)
        endforeach()
    endif()
endif()
